# Multi-stage build for Anchor contract & Keeper daemon
FROM node:20-slim AS keeper-builder
WORKDIR /app
COPY package.json tsconfig.json ./
RUN npm install
COPY agent/ ./agent/
RUN npm run build

FROM node:20-slim
WORKDIR /app
COPY --from=keeper-builder /app/package.json ./
COPY --from=keeper-builder /app/node_modules ./node_modules
COPY --from=keeper-builder /app/dist ./dist
ENV NODE_ENV=production
HEALTHCHECK --interval=30s --timeout=5s --start-period=5s --retries=3 \
  CMD node -e "require('http').get('http://localhost:8080/health', (r) => {if (r.statusCode !== 200) process.exit(1);})"
EXPOSE 8080
CMD ["node", "dist/index.js"]
